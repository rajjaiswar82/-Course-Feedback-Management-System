SECTION 8: QUICK REFERENCE - COMMON COMMANDS
================================================================================

# Build image
docker build -t course-feedback:1.0.0 .

# Run container (detached mode)
docker run -d --name feedback-app -p 8082:8080 -e APP_ENV=dev course-feedback:1.0.0

# Run container (foreground mode with logs)
docker run --name feedback-app -p 8082:8080 -e APP_ENV=dev course-feedback:1.0.0

# List running containers
docker ps

# List all containers (including stopped)
docker ps -a

# View container logs
docker logs feedback-app
docker logs -f feedback-app  # Follow logs (live stream)
docker logs --tail 50 feedback-app  # Last 50 lines

# Container lifecycle
docker stop feedback-app      # Stop gracefully
docker start feedback-app     # Start stopped container
docker restart feedback-app   # Restart running container
docker pause feedback-app     # Pause container
docker unpause feedback-app   # Unpause container

# Container inspection
docker inspect feedback-app   # Full JSON details
docker stats feedback-app     # Resource usage (live)
docker stats --no-stream feedback-app  # Resource usage (snapshot)
docker top feedback-app       # Running processes
docker port feedback-app      # Port mappings

# Execute commands in container
docker exec feedback-app ls -la /app
docker exec -it feedback-app sh  # Interactive shell

# Remove containers
docker rm feedback-app        # Remove stopped container
docker rm -f feedback-app     # Force remove running container
docker container prune        # Remove all stopped containers

# Remove images
docker rmi course-feedback:1.0.0
docker image prune            # Remove unused images

# Cleanup everything
docker system prune           # Remove stopped containers, unused networks, dangling images
docker system prune -a        # Remove all unused images (not just dangling)
